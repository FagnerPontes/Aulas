# LINUX
# Instalação (apenas se a criação falhar com erro de ensurepip):
sudo apt install python3-venv

# Criação do ambiente:
python3 -m venv venv

# Ativação:
source venv/bin/activate

# WINDOWS:
## Criar o ambiente
python -m venv venv

## Ativar no PowerShell
venv\Scripts\Activate.ps1

## Ativar no CMD
venv\Scripts\activate.bat
